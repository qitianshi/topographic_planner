cmake_minimum_required(VERSION 3.20)

project(pathfinder)

include_directories(.)
add_library(io io.c)
add_library(map map.c)
add_library(dyn_array adt/dynarr.c)
target_include_directories(dyn_array PRIVATE adt)
add_library(heap adt/heap.c)
target_include_directories(heap PRIVATE adt)
add_library(bitarr adt/bitarr.c)
target_include_directories(bitarr PRIVATE adt)
add_library(pathfinder pathfinder.c)
add_library(smooth smooth.c)
target_link_libraries(pathfinder heap smooth bitarr)
add_executable(extension main.c)
target_link_libraries(extension io map dyn_array heap pathfinder smooth m)

set (CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_SOURCE_DIR})
